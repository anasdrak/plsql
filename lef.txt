CREATE OR REPLACE PROCEDURE bar AS
BEGIN
    DBMS_OUTPUT.PUT_LINE('------------');
END;
/

CREATE OR REPLACE PROCEDURE title(j NUMBER) AS
BEGIN
    DBMS_OUTPUT.PUT_LINE('propaidia arithmou ' || j);
END;
/
CREATE OR REPLACE PROCEDURE propText(j number, i number ) as 
begin 
  DBMS_OUTPUT.PUT_LINE(j || ' x ' || i || ' = '|| (j*i));
end;
/
CREATE OR REPLACE PROCEDURE propaidia(k NUMBER) AS
BEGIN
    FOR j IN 0..k LOOP
        bar;
        title(j);
        bar;

        FOR i IN 0..10 LOOP
          propText(j,i);
        END LOOP;
    END LOOP;
END;
/

BEGIN
    propaidia(2);
END;
/
